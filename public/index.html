🛡️ Row Level Security (RLS) Ativo

Cada comerciante apenas acede aos seus próprios dados.

schema.sql 
-- LusaApp Builder — Schema SQL (Supabase/PostgreSQL)

CREATE TABLE public.profiles (
  id UUID PRIMARY KEY REFERENCES auth.users(id),
  email TEXT NOT NULL UNIQUE,
  business_name TEXT NOT NULL,
  nif TEXT CHECK (nif ~ '^[0-9]{9}$'),
  business_type TEXT CHECK (business_type IN ('restauracao','beleza','retalho','servicos')),
  postal_code TEXT CHECK (postal_code ~ '^[0-9]{4}-[0-9]{3}$'),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE public.posts (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  merchant_id UUID REFERENCES public.profiles(id),
  title TEXT NOT NULL,
  description TEXT NOT NULL,
  post_type TEXT CHECK (post_type IN ('promocao','noticia','evento')),
  published BOOLEAN DEFAULT false,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE public.customers (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  merchant_id UUID REFERENCES public.profiles(id),
  phone TEXT, email TEXT,
  loyalty_stamps INTEGER DEFAULT 0,
  rgpd_consent BOOLEAN DEFAULT false,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Row Level Security
ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY;
CREATE POLICY "merchant_own_profile" ON public.profiles 
  FOR SELECT USING (auth.uid() = id);
